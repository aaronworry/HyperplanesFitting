dim = 3                          # 数据维度
num = 20                         # 样本数量
hyperplanes_num = 6              # 每个样本的超平面数量 (增加)
max_distance_from_hyperplane = 0.2   # 点到超平面的最大噪声距离 (增加)
min_points_on_hyperplane = 20    # 每个超平面最少点数
max_points_on_hyperplane = 50    # 每个超平面最多点数 (不均匀)
X_limit = [-5., 5.]              # X 坐标范围
Y_limit = [-5., 5.]              # Y 坐标范围
Z_limit = [-5., 5.]              # Z 坐标范围

# 设计理由
# ---------
# 1. 增加超平面数量 (4 -> 6): 
#    - RANSAC 逐次拟合在后期会累积误差
#    - 流形优化的全局优化优势更明显
#
# 2. 增加噪声 (0.1 -> 0.2):
#    - RANSAC 的硬内点阈值容易失效
#    - 流形优化的软分配机制更鲁棒
#
# 3. 不均匀点数 (30-30 -> 20-50):
#    - RANSAC 对小模型容易漏检
#    - 流形优化的加权机制能更好处理
#
# 4. 超平面间距相应减小 (min_sep = 10 * 0.2 = 2.0):
#    - 超平面可能更接近，增加区分难度
#    - 有利于考察交叉区域的处理能力
